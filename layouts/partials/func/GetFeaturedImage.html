{{/* Return the featured image URL for a page and capture helpful scratch values. */}} {{- $page := . -}} {{- $angle := int (default 0 $page.Params.featured_image_angle) -}} {{- $page.Scratch.Set "featured_image_angle" $angle -}} {{- $page.Scratch.Delete "featured_image_resource" -}} {{- $page.Scratch.Delete "featuredAbsURL" -}} {{- $linkToCover := "" -}} {{- with $page.Params.featured_image -}} {{- $name := . -}} {{- $linkToCover = strings.Trim $name "/" | urls.AbsURL -}} {{- $candidates := slice $name (strings.TrimPrefix "/" $name) (strings.TrimPrefix "./" $name) -}} {{- range $candidate := $candidates -}} {{- if not ($page.Scratch.Get "featured_image_resource") -}} {{- with $page.Resources.GetMatch $candidate -}} {{- $page.Scratch.Set "featured_image_resource" . -}} {{- $page.Scratch.Set "featuredAbsURL" .Permalink -}} {{- $linkToCover = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- with $page.Resources.ByType "image" -}} {{- $matches := "feature,cover" -}} {{- with .GetMatch (fmt.Printf "**{%s}*" $matches) -}} {{- $page.Scratch.Set "featured_image_resource" . -}} {{- $page.Scratch.Set "featuredAbsURL" .Permalink -}} {{- $linkToCover = .RelPermalink -}} {{- else -}} {{- with .GetMatch "*" -}} {{- $page.Scratch.Set "featured_image_resource" . -}} {{- $page.Scratch.Set "featuredAbsURL" .Permalink -}} {{- $linkToCover = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (not ($page.Scratch.Get "featuredAbsURL")) $linkToCover -}} {{- $page.Scratch.Set "featuredAbsURL" $linkToCover -}} {{- end -}} {{- $linkToCover -}}
